def GetName():
    name = input("Entrez votre nom: ")
    return name


def GetRNDName():
    import random
    names = ["Alice", "Bob", "Charlie", "Diana"]
    return random.choice(names)


def print_hello(name_func):
    name = name_func()
    print(f"Bonjour, {name} !")


print_hello(GetName)
print_hello(GetRNDName)
